#!/usr/bin/env python3
"""Successful repair != correct root-cause diagnosis.

A representation-side compensator fixes one frozen query despite an upstream source fault.
A second query reveals the source remains wrong. Research-only construct.
"""
TRUTH={"A":10,"B":20}
FAULTY_SOURCE={"A":8,"B":20}

def baseline_query(q):
    if q == "A": return FAULTY_SOURCE["A"]
    if q == "SUM": return FAULTY_SOURCE["A"] + FAULTY_SOURCE["B"]
    raise KeyError(q)

def representation_compensator(q):
    # An overfit presentation/transform rule learned from the first target only.
    if q == "A": return FAULTY_SOURCE["A"] + 2
    if q == "SUM": return FAULTY_SOURCE["A"] + FAULTY_SOURCE["B"]
    raise KeyError(q)

def oracle(q):
    if q == "A": return TRUTH["A"]
    if q == "SUM": return TRUTH["A"] + TRUTH["B"]

print("# Repair-attribution counterexample\n")
for q in ("A","SUM"):
    b=baseline_query(q); r=representation_compensator(q); t=oracle(q)
    print(q,"baseline=",b,"representation-repair=",r,"truth=",t,"repair_correct=",r==t)

assert representation_compensator("A") == oracle("A")
assert representation_compensator("SUM") != oracle("SUM")
assert FAULTY_SOURCE != TRUTH
print("\nPASS: a representation intervention can compensate an upstream source fault on the measured target. Outcome improvement identifies treatment value for that target, not root-cause identity.")
